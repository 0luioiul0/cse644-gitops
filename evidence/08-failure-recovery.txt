$ ing web.gitops.local -o /dev/null -w 'web  -> HTTP %{http_code}\n' http://localhost:8090/

$ ing web.gitops.local http://localhost:8090/version
1.0.0

  # web answers, from the previous image. maxUnavailable: 0 refused to

  # remove a working replica for one that never became ready.

  # The 503 comes from the ingress controller, not from the application -

  # there is no application left to answer. An EndpointSlice is how the

  # controller learns where it may send a request, and readiness is what

  # gates each address in it:

$ kn get endpointslice -l kubernetes.io/service-name=api-clusterip -o custom-columns='SLICE:.metadata.name,ADDRESSES:.endpoints[*].addresses,READY:.endpoints[*].conditions.ready'
SLICE                 ADDRESSES       READY

$ kn get endpointslice -l kubernetes.io/service-name=web-clusterip -o custom-columns='SLICE:.metadata.name,ADDRESSES:.endpoints[*].addresses,READY:.endpoints[*].conditions.ready'
SLICE                 ADDRESSES                                                 READY
web-clusterip-j59w9   [10.244.1.32],[10.244.2.57],[10.244.2.56],[10.244.1.35]   true,true,true,false

  # web keeps three ready addresses and carries the stuck new Pod as a

  # fourth, not-ready one; the controller simply skips that address. api has

  # a single address and it is not ready, so there is nothing routable left.

  # That is the whole difference: RollingUpdate added a broken Pod alongside

  # working ones, Recreate deleted the working Pod before discovering the

  # replacement could not start.
